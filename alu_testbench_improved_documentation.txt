//============================================================
// 8-BIT ARITHMETIC LOGIC UNIT (ALU)
// Supports: ADD, SUB, AND, OR, XOR, NOT, SHIFT LEFT, SHIFT RIGHT
//============================================================
module alu_8bit(
    input wire [7:0] InputA,      // First 8-bit operand
    input wire [7:0] InputB,      // Second 8-bit operand
    input wire [2:0] OpCode,      // Operation selector
    output reg [15:0] OutALU,     // 16-bit result
    output reg COut               // Carry/borrow flag
);

    // Combinational logic for ALU operations
    always @(*) begin
        COut = 1'b0;  // Default: no carry

        case(OpCode)
            3'b000: begin  // ADDITION: OutALU = InputA + InputB
                {COut, OutALU} = InputA + InputB;
            end

            3'b001: begin  // SUBTRACTION: OutALU = InputA - InputB
                {COut, OutALU} = InputA - InputB;
            end

            3'b010: begin  // AND: OutALU = InputA & InputB
                OutALU = InputA & InputB;
            end

            3'b011: begin  // OR: OutALU = InputA | InputB
                OutALU = InputA | InputB;
            end

            3'b100: begin  // XOR: OutALU = InputA ^ InputB
                OutALU = InputA ^ InputB;
            end

            3'b101: begin  // NOT: OutALU = ~InputA
                OutALU = ~InputA;
            end

            3'b110: begin  // SHIFT LEFT: OutALU = InputA << 1
                OutALU = InputA << 1;
            end

            3'b111: begin  // SHIFT RIGHT: OutALU = InputA >> 1
                OutALU = InputA >> 1;
            end

            default: begin  // Undefined opcode
                OutALU = 16'b0;
                COut = 1'b0;
            end
        endcase
    end

endmodule
